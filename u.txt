CREATE TABLE strike_assets (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    armament VARCHAR(255) NOT NULL,
    max_fuel DECIMAL(10, 2) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE strikes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    strike_date DATETIME NOT NULL,
    target_id INT NOT NULL,
    strike_asset_id INT NOT NULL,
    location_id INT NOT NULL,
    result ENUM('missed', 'injured', 'dead') NOT NULL,
    CONSTRAINT fk_strike_target
        FOREIGN KEY (target_id) REFERENCES targets(id),
    CONSTRAINT fk_strike_asset
        FOREIGN KEY (strike_asset_id) REFERENCES strike_assets(id),
    CONSTRAINT fk_strike_location
        FOREIGN KEY (location_id) REFERENCES locations(id),
    CONSTRAINT uq_asset_target_location
        UNIQUE (strike_asset_id, target_id, location_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

def find_strike_counts(search_type, search_value):
    if search_type == "id":
        condition = "a.id = %s"
    elif search_type == "name":
        condition = "a.name = %s"
    else:
        raise ValueError("Search type must be id or name")

    query = f"""
        SELECT
            l.name,
            COUNT(DISTINCT t.id) AS target_count
        FROM strike_assets AS a
        JOIN strikes AS s ON s.strike_asset_id = a.id
        JOIN targets AS t ON t.id = s.target_id
        JOIN locations AS l ON l.id = s.location_id
        WHERE {condition}
        GROUP BY l.id, l.name
        ORDER BY l.id
    """

    with db.connection_context():
        cursor = db.execute_sql(query, (search_value,))
        rows = cursor.fetchall()

    results = []
    for row in rows:
        results.append({
            "location_name": row[0],
            "target_count": int(row[1]),
        })

    return results

@app.post("/strike-assets/investigation")
def investigate_strike_asset(request: StrikeRequest):
    if not check_user(request.username, request.password):
        raise HTTPException(
            status_code=401,
            detail="Invalid username or password",
        )

    if request.strike_asset_id is not None and request.name is not None:
        raise HTTPException(
            status_code=400,
            detail="Send either strike_asset_id or name, not both",
        )

    if request.strike_asset_id is not None:
        if request.strike_asset_id <= 0:
            raise HTTPException(
                status_code=400,
                detail="Strike asset ID must be greater than zero",
            )
        return find_strike_counts("id", request.strike_asset_id)

    if request.name is not None:
        name = request.name.strip()
        if not name:
            raise HTTPException(
                status_code=400,
                detail="Strike asset name cannot be empty",
            )
        return find_strike_counts("name", name)

    raise HTTPException(
        status_code=400,
        detail="Send strike_asset_id or name",
    )

def choose_action():
    return questionary.select(
        "Choose an action:",
        choices=[
            questionary.Choice("Last known location", value="location"),
            questionary.Choice("Register", value="register"),
            questionary.Choice("Strike asset investigation", value="investigation"),
            questionary.Choice("Exit", value="exit"),
        ],
    ).unsafe_ask()


def validate_text(value):
    if not value.strip():
        return "This field cannot be empty"
    return True


def read_credentials():
    username = questionary.text(
        "Enter username:",
        validate=validate_text,
    ).unsafe_ask()
    password = questionary.password(
        "Enter password:",
        validate=validate_text,
    ).unsafe_ask()

    return {
        "username": username.strip(),
        "password": password,
    }


def read_search_values():
    search_type = choose_search_type()
    if search_type == "id":
        search_value = read_target_id()
    else:
        search_value = read_target_name()
    return search_type, search_value

8. client/main.py — שורות 1 עד 38: להחליף את כל הקובץ.

import requests
from target_search import choose_action, read_credentials, read_search_values
from http_client import request_last_location, post_data
from output import show_last_location, show_strike_counts, show_message


def run_last_location():
    search_type, search_value = read_search_values()
    result = request_last_location(search_type, search_value)
    show_last_location(result)


def run_register():
    data = read_credentials()
    result = post_data("/register", data)
    show_message(result["message"])


def run_investigation():
    data = read_credentials()
    search_type, search_value = read_search_values()

    if search_type == "id":
        data["strike_asset_id"] = search_value
    else:
        data["name"] = search_value

    results = post_data("/strike-assets/investigation", data)
    show_strike_counts(results)


def main():
    while True:
        try:
            action = choose_action()

            if action == "exit":
                break
            if action == "location":
                run_last_location()
            elif action == "register":
                run_register()
            elif action == "investigation":
                run_investigation()

        except requests.exceptions.HTTPError as error:
            try:
                message = error.response.json().get("detail", str(error))
            except ValueError:
                message = str(error)
            show_message(f"Error: {message}")

        except requests.exceptions.RequestException as error:
            show_message(f"Connection error: {error}")

        except (KeyboardInterrupt, EOFError):
            break


def accepted_tp_cyber():
    while True:
        try:
            answer = input("did I got accepted to cyber? ")
        except (KeyboardInterrupt, EOFError):
            return
        if answer.strip().lower() == "yes":
            main()
            return
        print("wrong answer try again")


if __name__ == "__main__":
    accepted_tp_cyber()
