CREATE TABLE strike_assets (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    armament VARCHAR(255) NOT NULL,
    max_fuel DECIMAL(10, 2) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE strikes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    strike_date DATETIME NOT NULL,
    target_id INT NOT NULL,
    strike_asset_id INT NOT NULL,
    location_id INT NOT NULL,
    result ENUM('missed', 'injured', 'dead') NOT NULL,
    CONSTRAINT fk_strike_target
        FOREIGN KEY (target_id) REFERENCES targets(id),
    CONSTRAINT fk_strike_asset
        FOREIGN KEY (strike_asset_id) REFERENCES strike_assets(id),
    CONSTRAINT fk_strike_location
        FOREIGN KEY (location_id) REFERENCES locations(id),
    CONSTRAINT uq_asset_target_location
        UNIQUE (strike_asset_id, target_id, location_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


2. SQL/design.txt — שורה 1: להחליף את כל הקובץ.

strike_assets

name: VARCHAR(100), חובה וייחודי.
armament: VARCHAR(255), חובה.
max_fuel: DECIMAL(10,2), חובה.

strikes
id: INT AUTO_INCREMENT, מפתח ראשי.
strike_date: DATETIME, חובה.
target_id: INT, חובה, מפתח זר אל targets.id.
strike_asset_id: INT, חובה, מפתח זר אל strike_assets.id.
location_id: INT, חובה, מפתח זר אל locations.id.
result: ENUM('missed','injured','dead'), חובה.

השילוב strike_asset_id, target_id, location_id הוא ייחודי.
הטבלאות מקשרות כל תקיפה לאמצעי, למטרה ולמיקום התקיפה.
השאילתה מסננת לפי מזהה האמצעי או שמו, מקבצת לפי המיקום וסופרת COUNT(DISTINCT target_id).
מוחזר שם המיקום מתוך locations. הספירה כוללת את כל ערכי result, כולל missed.
משתמשים נשמרים בנפרד ב-server/users.csv בשדות username,password_hash.

import csv
from pathlib import Path
import bcrypt
from fastapi import HTTPException
USERS_FILE = Path(__file__).resolve().parent / "users.csv"

def read_users():
    if not USERS_FILE.exists():
        return []
    with USERS_FILE.open("r", encoding="utf-8", newline="") as file:
        users = list(csv.DictReader(file))
    return users

def register_user(username, password):
    username = username.strip()
    if not username or not password.strip():
        raise HTTPException(
            status_code=400,
            detail="Username and password cannot be empty",)
    password_bytes = password.encode("utf-8")

    if len(password_bytes) > 72:
        raise HTTPException(
            status_code=400,
            detail="Password is too long",)
    users = read_users()
    for user in users:
        if user["username"] == username:
            raise HTTPException(
                status_code=409,
                detail="Username already exists",)

    password_hash = bcrypt.hashpw(
        password_bytes,
        bcrypt.gensalt(),
    ).decode("utf-8")

    file_is_empty = (
        not USERS_FILE.exists()
        or USERS_FILE.stat().st_size == 0)

    with USERS_FILE.open("a", encoding="utf-8", newline="") as file:
        writer = csv.DictWriter(
            file,
            fieldnames=["username", "password_hash"],)
        if file_is_empty:
            writer.writeheader()
        writer.writerow({
            "username": username,
            "password_hash": password_hash,})
    return {"message": "User registered successfully"}

def check_user(username, password):
    username = username.strip()
    password_bytes = password.encode("utf-8")
    if not username or not password.strip():
        return False
    if len(password_bytes) > 72:
        return False
    users = read_users()
    for user in users:
        if user["username"] == username:
            saved_hash = user["password_hash"].encode("utf-8")
            password_is_correct = bcrypt.checkpw(
                password_bytes,
                saved_hash,)
            return password_is_correct
    return False


5. server/target_service.py — להוסיף בסוף הקובץ, אחרי find_last_location. הייבוא from db import db נשאר בראש הקובץ.

def find_strike_counts(search_type, search_value):
    if search_type == "id":
        condition = "a.id = %s"
    elif search_type == "name":
        condition = "a.name = %s"
    else:
        raise ValueError("Search type must be id or name")

    query = f"""
        SELECT
            l.name,
            COUNT(DISTINCT t.id) AS target_count
        FROM strike_assets AS a
        JOIN strikes AS s ON s.strike_asset_id = a.id
        JOIN targets AS t ON t.id = s.target_id
        JOIN locations AS l ON l.id = s.location_id
        WHERE {condition}
        GROUP BY l.id, l.name
        ORDER BY l.id
    """

    with db.connection_context():
        cursor = db.execute_sql(query, (search_value,))
        rows = cursor.fetchall()

    results = []
    for row in rows:
        results.append({
            "location_name": row[0],
            "target_count": int(row[1]),
        })

    return results

@app.post("/strike-assets/investigation")
def investigate_strike_asset(request: StrikeRequest):
    if not check_user(request.username, request.password):
        raise HTTPException(
            status_code=401,
            detail="Invalid username or password",
        )

    if request.strike_asset_id is not None and request.name is not None:
        raise HTTPException(
            status_code=400,
            detail="Send either strike_asset_id or name, not both",
        )

    if request.strike_asset_id is not None:
        if request.strike_asset_id <= 0:
            raise HTTPException(
                status_code=400,
                detail="Strike asset ID must be greater than zero",
            )
        return find_strike_counts("id", request.strike_asset_id)

    if request.name is not None:
        name = request.name.strip()
        if not name:
            raise HTTPException(
                status_code=400,
                detail="Strike asset name cannot be empty",
            )
        return find_strike_counts("name", name)

    raise HTTPException(
        status_code=400,
        detail="Send strike_asset_id or name",
    )

            questionary.Choice("Exit", value="exit"),

def validate_text(value):
    if not value.strip():
        return "This field cannot be empty"
    return True


def read_credentials():

10. client/main.py — שורה 1: להחליף את כל הקובץ.

import requests
from target_search import choose_action, read_credentials, read_search_values
from http_client import request_last_location, post_data
from output import show_last_location, show_strike_counts, show_message


def run_last_location():
    search_type, search_value = read_search_values()
    result = request_last_location(search_type, search_value)
    show_last_location(result)


def run_register():
    data = read_credentials()
    result = post_data("/register", data)
    show_message(result["message"])


def run_investigation():
    data = read_credentials()
    search_type, search_value = read_search_values()

    if search_type == "id":
        data["strike_asset_id"] = search_value
    else:
        data["name"] = search_value

    results = post_data("/strike-assets/investigation", data)
    show_strike_counts(results)


def main():
    while True:
        try:
            action = choose_action()

            if action == "exit":
                break
            if action == "location":
                run_last_location()
            elif action == "register":
                run_register()
            elif action == "investigation":
                run_investigation()

        except requests.exceptions.HTTPError as error:
            try:
                message = error.response.json().get("detail", str(error))
            except ValueError:
                message = str(error)
            show_message(f"Error: {message}")

        except requests.exceptions.RequestException as error:
            show_message(f"Connection error: {error}")

        except (KeyboardInterrupt, EOFError):
            break


if __name__ == "__main__":
    main()
